    Copyable patterns

    Audio recipe library

    Start with the smallest graph that fits your use case. Each snippet follows the same ownership rule: the context starts, suspends, and closes every nested source and recorder.

    MediaStream timing: audio-stream-output.stream becomes available after audio-context.resume() builds the graph.

    01 · INPUT TO OUTPUT

    Monitor one microphone

    Route a microphone directly to the selected context speaker.

    <audio-context id="audio">
  <audio-input-mic id="mic">
    <audio-output></audio-output>
  </audio-input-mic>
</audio-context>

<script>
  start.onclick = () => audio.resume();
</script>

    02 · FILE TO RECORDER

    Record a file without playback

    Connect a file only to a MediaStream-backed recorder.

    <audio-context id="audio">
  <audio-input-file src="./music.mp3">
    <audio-stream-output>
      <media-recorder timeslice="1000"></media-recorder>
    </audio-stream-output>
  </audio-input-file>
</audio-context>

    05 · DECLARATIVE EVENTS

    Handle recorder chunks in HTML

    Read each native Blob from the shared custom-event envelope.

    <audio-context id="audio">
  <audio-input-mic>
    <audio-stream-output>
      <media-recorder
        ondataavailable="HandleChunk(event)">
      </media-recorder>
    </audio-stream-output>
  </audio-input-mic>
</audio-context>

<script>
  function HandleChunk(event) {
    const blob = event.detail.data;
    console.log(blob, event.detail.metadata.timecode);
  }
</script>

    06 · DOWNLOAD

    Collect and download recording

    Request a chunk without stopping, then build a Blob after close.

    <audio-context id="audio">
  <audio-input-file src="./music.mp3">
    <audio-stream-output>
      <media-recorder id="recorder"></media-recorder>
    </audio-stream-output>
  </audio-input-file>
</audio-context>

<script>
  const chunks = [];
  recorder.addEventListener("dataavailable", (event) => {
    chunks.push(event.detail.data);
  });

  snapshot.onclick = () => recorder.requestData();
  finish.onclick = async () => {
    await audio.close();
    const recording = new Blob(chunks, { type: chunks[0]?.type });
    download.href = URL.createObjectURL(recording);
  };
</script>

    07 · NATIVE MEDIASTREAM

    Inspect generated audio tracks

    The stream is a standard browser MediaStream.

    <audio-context id="audio">
  <audio-input-mic>
    <audio-stream-output id="streamOutput">
      <media-recorder></media-recorder>
    </audio-stream-output>
  </audio-input-mic>
</audio-context>

<script>
  start.onclick = async () => {
    await audio.resume();
    const stream = streamOutput.stream;
    const [track] = stream.getAudioTracks();
    console.log(track.label, track.readyState);
  };
</script>

    08 · STANDARD API HANDOFF

    Send the graph to WebRTC

    Pass native tracks onward; signaling remains application-owned.

    <audio-context id="audio">
  <audio-input-mic>
    <audio-stream-output id="streamOutput">
      <media-recorder></media-recorder>
    </audio-stream-output>
  </audio-input-mic>
</audio-context>

<script>
  const peerConnection = new RTCPeerConnection();

  call.onclick = async () => {
    await audio.resume();
    const stream = streamOutput.stream;
    for (const track of stream.getAudioTracks()) {
      peerConnection.addTrack(track, stream);
    }
  };
</script>
